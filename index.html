// クロック描画のコアロジック function drawClock() { const canvas = document.getElementById('clockCanvas'); const ctx = canvas.getContext('2d'); const radius = canvas.width / 2; ctx.translate(radius, radius); // 1. 背景色の配置（最適化済み：上部ミントグリーン、下部ホワイト） // 上半分 ctx.beginPath(); ctx.arc(0, 0, radius * 0.95, Math.PI, 0); ctx.fillStyle = '#B2D8C1'; // ミントグリーン ctx.fill(); // 下半分 ctx.beginPath(); ctx.arc(0, 0, radius * 0.95, 0, Math.PI); ctx.fillStyle = '#FFFFFF'; // 白 ctx.fill(); // 2. 24時間表記のインデックス（短針用：1周24時間） draw24hNumbers(ctx, radius); // 3. 外周の60分メモリ（長針用） draw60mTicks(ctx, radius); // 4. ロゴ描画 drawLogo(ctx, radius); } function draw24hNumbers(ctx, radius) { ctx.font = radius * 0.1 + "px Arial"; ctx.fillStyle = "#333"; ctx.textAlign = "center"; ctx.textBaseline = "middle"; for (let num = 1; num <= 24; num++) { let ang = num * Math.PI / 12; ctx.rotate(ang); ctx.translate(0, -radius * 0.75); ctx.fillText(num.toString(), 0, 0); ctx.translate(0, radius * 0.75); ctx.rotate(-ang); } }
